{{- /* Order posts by issue, then by column sequence (X-AI ... X-Terms), so prev/next follows the reading order of each printed issue rather than raw publish date. */}} {{- $current := . }} {{- $allPages := where site.RegularPages "Type" "in" site.Params.mainSections }} {{- $allPages = where $allPages "Params.hiddenInHomeList" "!=" "true" }} {{- $columns := slice "X-AI" "X-Mindset" "X-Money" "X-Art" "X-Book" "X-Movie" "X-Terms" }} {{/* Collect distinct issue numbers, ascending */}} {{- $issues := slice }} {{- range $allPages }} {{- $n := .Params.issue | default 0 }} {{- if not (in $issues $n) }}{{- $issues = $issues | append $n }}{{- end }} {{- end }} {{- $issues = sort $issues }} {{- $ordered := slice }} {{- range $issues }} {{- $issuePosts := where $allPages "Params.issue" . }} {{- range $columns }} {{- $col := . }} {{- range $issuePosts }} {{- if in .Params.tags $col }} {{- $ordered = $ordered | append . }} {{- end }} {{- end }} {{- end }} {{- end }} {{- $pos := -1 }} {{- range $i, $p := $ordered }} {{- if eq $p $current }}{{- $pos = $i }}{{- end }} {{- end }} {{- if ge $pos 0 }}
